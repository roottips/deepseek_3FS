# =========================================================================
# Build-system additions for the 3FS GDR adaptation.
#
# Drop this fragment into the appropriate CMake target inside 3FS's
# existing build. The GDR sources (GpuMemMgr.{h,cc}) are added to the
# ibverbs library so RDMABuf.cc can use them; CUDA is linked lazily via
# dlopen (see GpuMemMgr.cc::CudaApi::get()), so there is no hard link
# dependency -- 3FS still builds on hosts without CUDA installed.
# =========================================================================

# --- 1. New sources for the ib library -----------------------------------
target_sources(hf3fs_ib PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/common/net/ib/GpuMemMgr.cc
)

# --- 2. Include paths ----------------------------------------------------
target_include_directories(hf3fs_ib PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/common/net/ib
)

# --- 3. CUDA driver + runtime headers (build-time only) ------------------
# We only need headers (the actual libcuda is dlopen'd at runtime). Use
# CUDAToolkit's header-only mode if available; otherwise fall back to
# system /usr/local/cuda/include.
find_package(CUDAToolkit QUIET)
if(CUDAToolkit_FOUND)
    target_include_directories(hf3fs_ib PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
else()
    target_include_directories(hf3fs_ib PRIVATE /usr/local/cuda/include)
endif()

# --- 4. Verbs dependency (already present; restated for clarity) ---------
# libibverbs must export ibv_reg_dmabuf_mr (>= 1.8 / rdma-core 33).
# Older Mellanox OFED 5.x also exposes it; MLNX_OFED 4.x does NOT.
# Detect at CMake time so the build fails fast on unsupported stacks.
include(CheckSymbolExists)
set(CMAKE_REQUIRED_LIBRARIES "ibverbs")
check_symbol_exists(ibv_reg_dmabuf_mr "infiniband/verbs.h"
                     HF3FS_HAS_IBV_REG_DMABUF_MR)
if(NOT HF3FS_HAS_IBV_REG_DMABUF_MR)
    message(WARNING
        "ibv_reg_dmabuf_mr not found in libibverbs. The GDR dma-buf path "
        "will be disabled; only the legacy nvidia-peermem path will work. "
        "Upgrade to rdma-core >= 33 or MLNX_OFED >= 5.x.")
    target_compile_definitions(hf3fs_ib PRIVATE HF3FS_NO_DMABUF_MR)
endif()

# --- 5. Feature flag for source-level disabling --------------------------
option(HF3FS_ENABLE_GDR "Build 3FS with GPUDirect RDMA support" ON)
if(HF3FS_ENABLE_GDR)
    target_compile_definitions(hf3fs_ib PUBLIC HF3FS_HAS_GDR=1)
endif()

# SGLang side: the sglang_gdr/ module is pure-Python and needs no CMake.
# Just drop it into python/sglang/srt/disaggregation/gdr/ and the existing
# setup.py picks it up automatically (it is a regular Python package).
#
# The one external runtime dependency is the `ibverbs` PyPI package
# (rdma4py). Document it in SGLang's setup.py extras_require so users
# can install with `pip install 'sglang[gdr]'`:
# =========================================================================
#
# In SGLang's pyproject.toml / setup.py, add to extras_require:
#
#     "gdr": ["ibverbs>=0.3.2"],
#
# And document the runtime prerequisites in the server_args docstring:
#     disaggregation_transfer_engine="gdr" requires:
#       - ibverbs (pip install ibverbs)
#       - NVIDIA driver R470+ with nvidia-peermem (modprobe nvidia-peermem)
#         OR CUDA 13+ for the dma-buf path (no kernel module needed)
#       - Mellanox ConnectX-6 Dx or later NIC on the same PCIe root
#         complex as the target GPU.
